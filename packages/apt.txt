# apt package list for Linux (Debian/Ubuntu) - parallels the brew lines in Brewfile.
#
# Line format:  name [@tag] [# comment]
#
# Blank lines and '#' comments are ignored. An optional tag restricts the line to
# machines it applies to; untagged lines apply everywhere:
#
#   @wsl      only under WSL
#   @!wsl     only outside WSL
#   @ubuntu   only on Ubuntu (not on plain Debian)
#
# This list is the only place package names live. setup/packages-linux.sh sets up
# the apt sources first (WakeMeOps, the helix PPA, NodeSource) and then installs
# everything here in one go. Only what apt cannot provide at all is installed
# separately there: antidote (git clone) and starship (upstream installer).

# Core / shell
zsh
tmux
coreutils
findutils
wget
rsync
tree
procps            # provides `watch`

# Prerequisites. Also installed before the apt sources are added — the source
# setup needs curl/gpg, and the antidote clone needs git — but listed here so the
# list stays the full picture.
git               # from the git-core PPA on Ubuntu (see setup/packages-linux.sh)
curl
ca-certificates
gnupg             # gpg
software-properties-common  # add-apt-repository, for the git and helix PPAs

# Editors
vim
helix @ubuntu     # PPA (see setup/packages-linux.sh); no Debian package exists

# Python
python3
python3-venv
python3-pip
ipython3

# Dev tooling
universal-ctags   # ctags
httpie
nodejs            # NodeSource repo, added by setup/packages-linux.sh

# Kubernetes / ops (WakeMeOps repo — none of these are packaged by Debian)
kubectl
helm
yq                # mikefarah's Go yq; Debian's `yq` is the unrelated python jq wrapper

# Networking
nmap
mtr-tiny          # mtr (the `mtr` metapackage pulls in GTK)
wrk               # in universe on Ubuntu (not packaged on Debian)

# Clipboard, for the pbcopy/pbpaste shims in home/.functions. Under WSL the shims
# go through win32yank.exe/clip.exe instead, so the X11/Wayland tooling there would
# only drag in dependencies nothing ever calls.
xclip        @!wsl
wl-clipboard @!wsl

# Misc
bat               # from WakeMeOps as `bat`; Debian's own package installs `batcat`
wslu @wsl         # wslview, used by the `open` shim in home/.functions
